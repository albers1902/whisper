+if [ -z "$CIH" ]; then iso="CI-brugeren $CIU findes ikke"
+elif ! sudo -n -u "$CIU" true 2>/dev/null; then iso="sudo-reglen for $CIU mangler"
+else
+  while IFS= read -r f; do sudo -n -u "$CIU" test -r "$HOME_DIR/$f" && iso="$CIU kan læse $f"; done \
+    < <(find config private .my.cnf .git-credentials .ssh -type f 2>/dev/null || true)
+  case "$(cfgs ci_database pass)" in *'"'*|*'\'*) iso="${iso:-config.ini [ci_database] pass må ikke indeholde \" eller \\}" ;; esac
+  sudo -n -u "$CIU" test -x "$TOOLS" || iso="${iso:-$CIU kan ikke nå $TOOLS (chmod 711 $HOME_DIR)}"
+  [ -z "$CW_COMPOSER" ] || sudo -n -u "$CIU" test -r "$CW_COMPOSER" || iso="${iso:-$CIU kan ikke læse composer ($CW_COMPOSER) — composer audit kan ikke køre}"
+fi
+if [ -n "$iso" ]; then
+  grep -qx "$SHA" "$CI/isofail" 2>/dev/null || { status failure "CI kører ikke: $iso (OPSAETNING.md: CI-bruger)"; echo "$SHA" >> "$CI/isofail"; }
+  sed -i "\|^$CTX $SHA$|d" "$CI/attempts"
+  alert "CI kører ikke: $iso. Opret CI-brugeren (OPSAETNING.md, Serveropsætning) — indtil da testes intet."
+  exit 0
+fi
+
+LOG="$CI/runs/${SHA:0:12}-$KIND.log"; : > "$LOG"
+find "$CI/runs" -name '*.log' -mtime +30 -delete 2>/dev/null || true
+echo "$(date -Is) ci: $CTX ${SHA:0:12} ${REF:+(PR #$PRN $REF)} start"
+APIW "statuses/$SHA" "$(body pending "kører på DEV-serveren")" || true
+if [ "$KIND" = pr ]; then BASE=$(WG merge-base origin/main "$SHA")
+else BASE=$(WG rev-parse -q --verify "$SHA^1" 2>/dev/null || true); fi
+INI=$(WG show "$SHA:coworker.ini" 2>/dev/null || true)     # projektets værdier fra den testede commit (kontrolfil)
+CW_TESTS=$(ini ci tests); CW_ANALYSE=$(ini ci analyse); CW_PHPSTAN=$(ini ci phpstan); CW_MIGBASE=$(ini migrations baseline)
+# [ci] phpstan fra den testede commit vælger aldrig værktøj eller sti: kun den låste version er tilladt
+if [ -n "$CW_PHPSTAN" ] && [ "$CW_PHPSTAN" != "$PHPSTAN_V" ]; then CW_PHPSTAN_BAD=1; else CW_PHPSTAN_BAD=""; fi
+CW_PHPSTAN=$PHPSTAN_V
+FAILS=()
+
+# Tunge trin kun ved kode/test/kontrolfiler; genbrug, hvis den samme kode allerede var grøn (fx PR'en lige før merge,
+# eller kun dokumentation lagt oven på en grøn commit — 5.26). "Koden" = alle filer undtagen *.md; de tunge trin
+# (lint, PHPStan, Semgrep, audit, migrationer, tests) læser ikke markdown. Gitleaks og report.sh ci kører altid.
+TREE=$(WG ls-tree -r "$SHA" | grep -v $'\t.*\\.md$' | sha256sum | cut -c1-40)
+HEAVY=true
+if [ -n "$BASE" ]; then
+  if WG diff --quiet "$BASE" "$SHA" -- homedir tests && WG diff --quiet "$BASE" "$SHA" -- "${CTRL[@]}"; then HEAVY=false; fi
+  if [ $HEAVY = true ] && WG diff --quiet "$BASE" "$SHA" -- "${CTRL[@]}" && grep -qE " $TREE success heavy$" "$CI/done" 2>/dev/null; then
+    HEAVY=false; echo "== tunge trin genbrugt: den samme kode (${TREE:0:12}, alt undtagen *.md) var grøn før" >> "$LOG"; fi
+fi
+
+# ---------- Trinene kører som CI-brugeren (ingen adgang til DEV's config/token). Scriptet lægges frisk hver gang ----------
+# CI-brugeren nulstilles før og efter hvert job: alle dens processer dræbes, og dens hjemmemappe tømmes — intet fra én PR
+# kan påvirke et senere job. Runneren ligger i ci-tools/ (DEV ejer den), så CI-brugeren kan ikke ændre den.
+ci_reset() { sudo -n -u "$CIU" sh -c 'kill -9 -1 2>/dev/null; find "$HOME" -mindepth 1 -delete 2>/dev/null; mkdir -p "$HOME/tmp"; true' 2>/dev/null || true; }
+asci() { sudo -n -u "$CIU" env -i PATH=/usr/local/bin:/usr/bin:/bin HOME="$CIH" TMPDIR="$CIH/tmp" LANG=C.UTF-8 W="$CIH/work" TOOLS="$TOOLS" \
+           GIT_CONFIG_GLOBAL=/dev/null GIT_CONFIG_NOSYSTEM=1 \
+           SHA="$SHA" BASE="$BASE" KIND="$KIND" REF="$REF" PRN="$PRN" PHP="$PHP" CW_COMPOSER="$CW_COMPOSER" \
+           CW_TESTS="$CW_TESTS" CW_ANALYSE="$CW_ANALYSE" CW_PHPSTAN="$CW_PHPSTAN" CW_PHPSTAN_BAD="$CW_PHPSTAN_BAD" CW_MIGBASE="$CW_MIGBASE" \
+           GITLEAKS_V="$GITLEAKS_V" SEMGREP_V="$SEMGREP_V" bash "$TOOLS/runner.sh" "$@"; }
+step() {   # $1=navn $2=trin i runneren. Output til loggen (højst 20 MB pr. trin); fejl huskes, næste trin kører alligevel
+  local n=$1 rc; echo "== $n" >> "$LOG"
+  set +e; asci "$2" 2>&1 9>&- | head -c 20000000 >> "$LOG"; rc=${PIPESTATUS[0]}; set -e
+  if [ "$rc" = 0 ]; then echo "OK" >> "$LOG"
+  elif [ "$2" = audit ] && [ "$rc" = 3 ]; then echo "RØD: $n — kunne ikke køre (ikke en sårbarhed)" >> "$LOG"; FAILS+=("$n — kunne ikke køre")
+  elif [ "$2" = audit ]; then echo "RØD: $n — sårbarhed fundet" >> "$LOG"; FAILS+=("$n — sårbarhed fundet")
+  else echo "RØD: $n" >> "$LOG"; FAILS+=("$n"); fi
+}
+ci_reset
+cat > "$TOOLS/runner.sh" <<'RUNNER'
+set -uo pipefail
+db() { awk -F= -v k="$1" '$1==k {sub(/^[^=]*=/,""); gsub(/^"|"$/,""); print; exit}' "$HOME/my.cnf"; }
+DBNAME=$(db database)
+case "$1" in
+  prep)   # frisk kopi af koden (alle refs, så critic-reviewets commit og base findes)
+    rm -rf "$W" && git init -q "$W" && git -C "$W" fetch -q --update-head-ok "$HOME/job.bundle" '+refs/*:refs/*' \
+      && git -C "$W" -c advice.detachedHead=false checkout -q -f --detach "$SHA" ;;
+  gitleaks) cd "$W"; cf=(); [ -f shared/gitleaks.toml ] && cf=(--config shared/gitleaks.toml)
+    "$TOOLS/gitleaks-$GITLEAKS_V/gitleaks" git --redact --no-banner "${cf[@]}" ${BASE:+--log-opts="$BASE..$SHA"} . ;;
+  lint) cd "$W"; find homedir -name '*.php' -print0 2>/dev/null | xargs -0 -r -n1 "$PHP" -l > /dev/null ;;
+  phpstan) cd "$W"; p="$TOOLS/phpstan-$CW_PHPSTAN/phpstan.phar"
+    [ -z "$CW_PHPSTAN_BAD" ] || { echo "coworker.ini [ci] phpstan: kun $CW_PHPSTAN er hash-låst i frameworket — fjern linjen"; exit 1; }
+    [ -f "$p" ] || { echo "PHPStan $CW_PHPSTAN er ikke hentet (cron/ci.sh tools)"; exit 1; }
+    [ -n "$CW_ANALYSE" ] || { echo "coworker.ini mangler [ci] analyse"; exit 1; }
+    # Egen tmpDir i CI-brugerens hjemmemappe (tømmes før og efter hvert job). Standarden /tmp/phpstan deles af alle
+    # brugere på serveren: ejes af den første (de andre fejler), og cachen er PHP-kode, PHPStan indlæser (5.21).
+    mkdir -p "$HOME/phpstan-tmp"
+    { printf 'parameters:\n\ttmpDir: %s\n' "$HOME/phpstan-tmp"
+      for f in phpstan-baseline.neon phpstan.neon phpstan.neon.dist; do   # som før: baseline, ellers PHPStans egen config
+        if [ -f "$f" ]; then printf 'includes:\n\t- %s\n' "$W/$f"; break; fi; done; } > "$HOME/phpstan-ci.neon"
+    echo "PHPStan $CW_PHPSTAN (sha256-låst phar), config:"; sed 's/^/  /' "$HOME/phpstan-ci.neon"
+    "$PHP" -d memory_limit=1G "$p" analyse --level=5 --no-progress -c "$HOME/phpstan-ci.neon" $CW_ANALYSE ;;
+  semgrep) cd "$W"; SEMGREP_ENABLE_VERSION_CHECK=0 "$TOOLS/semgrep-$SEMGREP_V/bin/semgrep" scan --config p/php --config p/secrets --error --metrics=off --quiet homedir/ ;;
+  audit) cd "$W"; [ -f homedir/composer.lock ] || { echo "ingen composer.lock"; exit 0; }
+    # En lock uden pakker har intet at auditere, men composer svarer rødt på den (versionsafhængigt) — U2. Kun et gyldigt
+    # JSON-objekt, hvor packages og packages-dev er tomme lister eller fraværende, springer over; alt andet afgør composer.
+    "$PHP" -r '$j = json_decode((string) @file_get_contents($argv[1]));
+      if (!is_object($j)) exit(1);
+      foreach (["packages", "packages-dev"] as $k) if (property_exists($j, $k) && $j->{$k} !== []) exit(1);
+      exit(0);' homedir/composer.lock 2> /dev/null && { echo "composer.lock har ingen pakker — audit sprunget over"; exit 0; }
+    # exit 1 = sårbarhed fundet, exit 3 = audit kunne ikke køre (miljø, netværk, composer) — aldrig det samme (5.26)
+    [ -n "$CW_COMPOSER" ] && [ -r "$CW_COMPOSER" ] || { echo "AUDIT KUNNE IKKE KØRE: composer findes ikke eller kan ikke læses af CI-brugeren ('$CW_COMPOSER')"; exit 3; }
+    cd homedir; rc=0
+    env -u COMPOSER "$PHP" "$CW_COMPOSER" audit --locked --no-interaction --format=json > "$HOME/audit.json" 2> "$HOME/audit.err" || rc=$?
+    "$PHP" -r '$j = json_decode((string) @file_get_contents($argv[1]), true);
+      if (!is_array($j) || !array_key_exists("advisories", $j)) exit(3);
+      $n = 0; foreach ((array) $j["advisories"] as $p => $l) foreach ((array) $l as $a) { $n++;
+        echo "SÅRBARHED: $p ", $a["advisoryId"] ?? "", " ", $a["title"] ?? "", " (berørte versioner: ", $a["affectedVersions"] ?? "?", ")\n"; }
+      foreach ((array) ($j["abandoned"] ?? []) as $p => $r) echo "Forladt pakke (ikke rødt): $p", $r ? " → $r" : "", "\n";
+      echo "$n sårbarhed(er)\n"; exit($n ? 1 : 0);' "$HOME/audit.json"; v=$?
+    [ "$v" = 3 ] && echo "AUDIT KUNNE IKKE KØRE (composer exit $rc): $(head -c 600 "$HOME/audit.err" | tr '\n' ' ')"
+    exit "$v" ;;
+  db) mysql --defaults-extra-file="$HOME/my.cnf" -e "DROP DATABASE IF EXISTS \`$DBNAME\`; CREATE DATABASE \`$DBNAME\` CHARACTER SET utf8mb4 COLLATE utf8mb4_unicode_ci" ;;
+  migrations) cd "$W"; DB=(mysql --defaults-extra-file="$HOME/my.cnf" "$DBNAME"); NEW=()
+    shopt -s nullglob; ALL=(homedir/migrations/[0-9]*.sql); shopt -u nullglob
